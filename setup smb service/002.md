```shell

# install samba server on ubuntu
sudo apt update
sudo apt install samba -y


# create or choose your user
sudo adduser admin_gtr
# password: gtr_server

# set samba password for the user
sudo smbpasswd -a admin_gtr
# it will prompt you to enter and confirm the password
# password: gtr_server

# enable the user
sudo smbpasswd -e admin_gtr


# [OPTIONAL]: change smb password for the user
sudo smbpasswd admin_gtr


# prepare your shared folder
sudo mkdir -p /mnt/storage/admin_gtr
sudo chown admin_gtr:admin_gtr /mnt/storage/admin_gtr
sudo chmod 770 /mnt/storage/admin_gtr

sudo chown $USER:$USER /home/$USER

# configure Samba share
sudo nano /etc/samba/smb.conf


# At the bottom, add this block:
[Home]
   path = /home/msl
   browseable = yes
   read only = no
   valid users = msl
   create mask = 0770
   directory mask = 0770
   force user = msl
   force group = msl
# You can change `[smbshare]` to any share name (e.g., `[storage]`).



# restart Samba service and check status
sudo systemctl daemon-reload
sudo systemctl restart smbd
sudo systemctl status smbd


# set maximum storage quota for admin_gtr to 100MB
# sudo apt install quota -y
# sudo mount -o remount /mnt/storage
# sudo quotacheck -cum /mnt/storage
# sudo quotaon -v /mnt/storage

# set quota for user admin_gtr to 1TB
sudo setquota -u admin_gtr 1000000000 1000000000 0 0 /mnt/storage
# 1TB = 1,000,000,000 blocks (assuming 1 block = 1KB)
# Adjust block count if your filesystem uses a different block size.


# [OPTIONAL] allow Samba through UFW firewall
sudo ufw allow samba

```

## Windows:

1. Open **File Explorer**
2. In the address bar, type:

   \\your_server_ip\smbshare

3. Log in with:

   - **Username:** admin_gtr
   - **Password:** (the one you set via `smbpasswd`)

## Linux:

smbclient //your_server_ip/smbshare -U admin_gtr
