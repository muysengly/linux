```bash

#
sudo apt update && sudo apt upgrade -y

# add a new user
sudo adduser production


# create new group
sudo groupadd product


# set production user as sudoer
sudo usermod -aG sudo production

# add user to specific group
sudo usermod -aG product product
# `-aG`: append user to group



# add user password
# sudo passwd [PASSWORD]
sudo passwd aaa

# switch to group
su - students

# switch to new user
su - aaa
# `-` is used to load the user's environment variables

# check user password
sudo -i -u product

# view current user password
whoami

#  set new password from root
sudo passwd product



# set user to admin mode
sudo usermod -aG admin aaa


```

# Remove USER Completely

```bash

# show all user
cut -d: -f1 /etc/passwd

# kill all processes of user before deletion
sudo pkill -u product


# remove user and their home directory
# ensure no running processes or sessions for 'product'
sudo ps -u product -o pid,comm
sudo kill -9 4949 || true
sudo pkill -u product || true
sudo loginctl terminate-user product || true
sudo systemctl --type=service --all | grep product || true

# now remove user and their home directory
sudo userdel -r product
# `-r`: removes home directory and mail spool

# verify user is deleted
id deployer
# should return "no such user"

# remove user from all groups (if needed)
sudo gpasswd -d aaa students

# check if user's processes are still running
ps -u aaa

# kill all user's processes (if any)
sudo pkill -u aaa

# verify user's cron jobs are removed
sudo crontab -u aaa -l

# remove user's cron jobs (if any)
sudo crontab -u aaa -r

# show all users
cut -d: -f1 /etc/passwd


```
