# Setup Service on Ubuntu

```shell

# create GTR service
sudo nano /etc/systemd/system/msl_backend.service


[Unit]
Description='MSL Backend Service'
After=network.target

[Service]
User=root
WorkingDirectory=/home/msl/msl_backend/
ExecStart=/bin/bash -c 'source /home/msl/msl_backend/venv/bin/activate && python service_main.py'
Restart=always

[Install]
WantedBy=multi-user.target




# hypercorn main:app --bind 0.0.0.0:8000 --workers 4

# reload the systemd daemon to recognize the new service file
sudo systemctl daemon-reexec
sudo systemctl daemon-reload

# enable and start a service
sudo systemctl enable toblork.service
sudo systemctl start toblork.service
sudo systemctl status toblork.service

# reload toblork.service service
sudo systemctl restart toblork.service
sudo systemctl status toblork.service
# stop toblork.service service
sudo systemctl disable toblork.service
sudo systemctl stop toblork.service

# remove toblork.service service
sudo rm -rf /etc/systemd/system/toblork.service.service

# check system memory
free -h

```

```bash

########## Complete Script to Create Service ##########

# variables
FILE_NAME="toblork.service"
WORKING_DIR="/home/msl/toblork_backend/"
ACTIVATE_VENV="source /home/msl/toblork_backend/venv/bin/activate"
RUN_COMMAND="python run_main.py"

# script
FILE_PATH="/etc/systemd/system/${FILE_NAME}"
sudo tee ${FILE_PATH} > /dev/null <<EOF
[Unit]
Description='My Service'
After=network.target

[Service]
User=root
WorkingDirectory=${WORKING_DIR}
ExecStart=/bin/bash -c '${ACTIVATE_VENV} && ${RUN_COMMAND}'
Restart=always

[Install]
WantedBy=multi-user.target
EOF

# reload the systemd daemon
sudo systemctl daemon-reexec
sudo systemctl daemon-reload

# enable and start service
sudo systemctl enable ${FILE_NAME}
sudo systemctl start ${FILE_NAME}
sudo systemctl status ${FILE_NAME}
```
