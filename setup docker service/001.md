# Setup Docker on Ubuntu

```shell

# update and upgrade packages
apt update && apt upgrade -y


# install docker
apt install docker.io -y


```

---

## Example:

```shell

# download latest mongodb docker image
docker pull mongo:latest

# list all docker images
docker images

# remove mongodb docker image
docker rmi mongo:latest


# run mongodb docker container with authentication

docker run -d `
    --restart unless-stopped `
    --name mongo_gtr `
    -p 27018:27017 `
    -e MONGO_INITDB_ROOT_USERNAME=admin `
    -e MONGO_INITDB_ROOT_PASSWORD="admin" `
    mongo:latest


docker run -d \
    --restart unless-stopped \
    --name mongo_production \
    -p 27017:27017 \
    -e MONGO_INITDB_ROOT_USERNAME=admin \
    -e MONGO_INITDB_ROOT_PASSWORD="asdfghjkl;'" \
    mongo:latest

# 27018 is the machine port
# 27017 is the container port
# MONGO_INITDB_ROOT_USERNAME is the admin username
# MONGO_INITDB_ROOT_PASSWORD is the admin password


# list all docker containers
docker ps -a


# stop and remove mongodb docker container
docker stop mongo_dev
docker rm mongo_dev


```

# Remove Docker Completely from Ubuntu

```bash
## Step 1: Stop All Docker Containers and Services
# Stop all running containers
sudo docker stop $(sudo docker ps -aq) 2>/dev/null
# Stop Docker services
sudo systemctl stop docker
sudo systemctl stop docker.socket
sudo systemctl stop containerd


## Step 2: Remove Docker Packages
# Remove Docker packages
sudo apt purge docker.io docker-doc docker-compose docker-compose-v2 podman-docker containerd runc -y
# Remove any other Docker-related packages
sudo apt purge docker* containerd* runc* -y
# Remove unused dependencies
sudo apt autoremove -y


## Step 3: Remove Docker Images, Containers, and Volumes
# Remove all Docker containers, images, volumes, and networks
sudo rm -rf /var/lib/docker
sudo rm -rf /var/lib/containerd
sudo rm -rf /var/lib/docker-engine


## Step 4: Remove Docker Configuration and Logs
# Remove configuration files
sudo rm -rf /etc/docker
sudo rm -rf /etc/containerd
sudo rm -rf ~/.docker
# Remove log files
sudo rm -rf /var/log/docker
sudo rm -rf /var/log/containerd


## Step 5: Remove Docker Systemd Services
# Disable and remove systemd services
sudo systemctl disable docker
sudo systemctl disable docker.socket
sudo systemctl disable containerd
# Remove systemd service files (if they exist)
sudo rm -f /etc/systemd/system/docker.service
sudo rm -f /etc/systemd/system/docker.socket
sudo rm -f /lib/systemd/system/docker.service
sudo rm -f /lib/systemd/system/docker.socket
sudo rm -f /lib/systemd/system/containerd.service
# Reload systemd
sudo systemctl daemon-reload
sudo systemctl reset-failed


## Step 6: Remove Docker User Groups and Users
# Remove docker user from groups (optional)
sudo gpasswd -d $USER docker 2>/dev/null
# Remove docker group (be careful with this)
sudo groupdel docker 2>/dev/null


## Step 7: Clean Up APT Repository (if Docker was installed via repository)
# Remove Docker repository
sudo rm -f /etc/apt/sources.list.d/docker.list
# Remove Docker GPG key
sudo rm -f /etc/apt/trusted.gpg.d/docker.gpg
sudo rm -f /usr/share/keyrings/docker-archive-keyring.gpg
# Update package list
sudo apt update


## Step 8: Verify Complete Removal
# Check if Docker packages are removed
dpkg -l | grep -E "docker|containerd"
# Check if Docker processes are running
ps aux | grep -E "docker|containerd"
# Check if Docker directories still exist
ls -la /var/lib/ | grep -E "docker|containerd"
ls -la /etc/ | grep -E "docker|containerd"
# Test docker command (should show "command not found")
docker --version


## Step 9: Remove Any Leftover Files
# Search for any remaining Docker files
sudo find / -name "*docker*" -type f 2>/dev/null
sudo find / -name "*containerd*" -type f 2>/dev/null
# Remove any found files (be careful with this command)
# sudo find / -name "*docker*" -type f -delete 2>/dev/null
# sudo find / -name "*containerd*" -type f -delete 2>/dev/null


## Step 10: Clean APT Cache
# Clean package cache
sudo apt clean
sudo apt autoclean
# Update package list
sudo apt update

## Final Verification Commands
# Verify Docker is completely removed
which docker
docker --version
systemctl status docker
# Check for any remaining processes
pgrep -f docker
pgrep -f containerd
# Check installed packages
apt list --installed | grep -E "docker|containerd"
```
