# Setup Docker on Ubuntu

```shell

# update and upgrade packages
sudo apt update && sudo apt upgrade -y


# install docker
sudo apt install docker.io -y


# reload docker service
sudo systemctl daemon-reexec
sudo systemctl daemon-reload

# enable docker service
sudo systemctl enable docker
sudo systemctl start docker
sudo systemctl status docker


# pull lastest mongodb docker image
sudo docker pull mongo:latest

# list all docker images
sudo docker images


# run mongodb docker container with authentication
sudo

docker run -d `
    -p 127.0.0.1:27017:27017 `
    --name mongo_product `
    -e MONGO_INITDB_ROOT_USERNAME=admin_pro `
    -e MONGO_INITDB_ROOT_PASSWORD=admin_pro `
    mongo:latest



sudo docker run -d `
    -p 127.0.0.1:27018:27017 `
    --name mongo_developer `
    -e MONGO_INITDB_ROOT_USERNAME=admin_dev `
    -e MONGO_INITDB_ROOT_PASSWORD=admin_dev `
    mongo:latest



# list all docker containers
sudo docker ps -a


sudo docker stop mongo_product
sudo docker rm mongo_product

sudo docker stop mongo_developer
sudo docker rm mongo_developer



cloudflared access tcp --hostname dev.codeshift.me --url 127.0.0.1:27018






# set docker access permission for current user
sudo usermod -aG docker $USER





# run mongo shell to connect to mongo_dev container
sudo docker exec -it mongo_dev mongo


# DEMO setup mongodb for raspberry pi via docker

# pull compatible mongo docker image (≤4.4.18 for ARM systems)
sudo docker pull mongo:4.4.18

# list all docker images
sudo docker images

# run docker container with compatible version
sudo docker run -d -p 27017:27017 --name mongo_container mongo:4.4.18


sudo docker run -d -p 127.0.0.1:27017:27017 --name mongo_container1 mongo:4.4.18

# set mongo_container access username and password
# default: no authentication
# to enable authentication, create admin user in mongo shell
# then restart mongo container with authentication options
# example to create admin user:
sudo docker exec -it mongo_container mongo
use admin
db.createUser({
  user: "admin",
  pwd: "password",
    roles: [ { role: "userAdminAnyDatabase", db: "admin" } ]
})
exit
# restart mongo container with authentication
sudo docker stop mongo_container
sudo docker rm mongo_container
sudo docker run -d -p 27017:27017 --name mongo_container \
  mongo:4.4.18 --authenticationDatabase "admin"
# now connect with username and password
sudo docker exec -it mongo_container mongo -u "admin" -p "password" --authenticationDatabase "admin"


# list all docker containers
sudo docker ps -a

# goto mongo docker container bash shell
sudo docker exec -it mongo_container /bin/bash

# check mongo docker container logs
sudo docker logs mongo_container

# stop mongo docker container
sudo docker stop mongo_container

# start mongo docker container
sudo docker start mongo_container

# remove mongo docker container
sudo docker rm -f mongo_container

# remove all docker containers
sudo docker rm -f $(sudo docker ps -a -q)


# pull fastapi docker image
sudo docker pull tiangolo/uvicorn-gunicorn-fastapi:latest

# pull fastapi docker image without tag
sudo docker pull tiangolo/uvicorn-gunicorn-fastapi

# run fastapi docker container
sudo docker run -d -p 8000:80 --name fastapi_container tiangolo/uvicorn-gunicorn-fastapi:latest
# 8000 on host maps to 80 in container

# add port mapping for fastapi_container container
sudo docker run -d -p 2222:22 -p 8000:80 --name aaa tiangolo/uvicorn-gunicorn-fastapi:latest

# container source code path
# /app


# ssh to docker container
sudo docker exec -it aaa /bin/bash

# add ssh to fastapi container
apt-get update
apt-get install openssh-server -y
service ssh start



# set root password
passwd
# NOTE: use this password to login via ssh
# check ssh service status
service ssh status

# login to fastapi container via ssh
ssh root@localhost -p 2222

# forward port 2222 on host to port 22 in container
sudo docker run -d -p 2222:22 --name fastapi_container_with_ssh tiangolo/uvicorn-gunicorn-fastapi:latest
```
