# Setup Ubuntu System

```shell
# open teletypewriter
# tty: ctrl + alt + F3
# return to GUI: ctrl + alt + F2

# update and upgrade packages
sudo apt update && sudo apt upgrade -y

# autoremove unnecessary packages
sudo apt autoremove -y

# full upgrade packages for kernel updates
sudo apt-get full-upgrade -y
sudo reboot

# install necessary packages
sudo apt install curl -y







# [OPTIONAL]: create a new user in mnt/storage/home
sudo mkdir -p /mnt/storage/home
sudo useradd -d /mnt/storage/home/test_user -m -s /bin/bash test_user

# login as root
sudo -i

# create a new user with home directory
# sudo adduser [USER]
sudo adduser test_user

# add user to sudo group
sudo usermod -aG sudo test_user

# add user password
# sudo passwd [PASSWORD]
sudo passwd test_user

# switch to new user
su - test_user
su - gtr


# change user password
passwd


# add access to mysql with specific database
sudo usermod -aG mysql test_user


# rename user
sudo usermod -l new_username old_username
> sudo usermod -l new_test_user test_user
# -l: new login name


# list all users
awk -F: '$3>=1000 {print $1}' /etc/passwd
# or
cat /etc/passwd


# [DELETE USER]: remove user and all associated files
# open new terminal

# check processes owned by the user
ps -u test_user
# kill all processes owned by the user
sudo pkill -u test_user
# delete user
sudo deluser test_user
# delete user along with home directory
sudo deluser --remove-home test_user
# delete all files owned by the user
sudo find / -user test_user -exec rm -rf {} \;
# delete user mail spool
sudo rm -f /var/mail/test_user




```

```shell
# configure ubuntu power settings (no sleep/suspend)
# disable automatic suspend
sudo systemctl mask sleep.target suspend.target hibernate.target hybrid-sleep.target

# modify logind.conf to ignore lid switch
sudo sed -i 's/#HandleLidSwitch=suspend/HandleLidSwitch=ignore/' /etc/systemd/logind.conf
sudo sed -i 's/#HandleLidSwitchExternalPower=suspend/HandleLidSwitchExternalPower=ignore/' /etc/systemd/logind.conf

# restart systemd-logind service
sudo systemctl restart systemd-logind

# verify power settings
systemctl status sleep.target


# Manually:

# sudo systemctl restart systemd-logind

# rename ubuntu server hostname
sudo hostnamectl set-hostname gtr-server



```
