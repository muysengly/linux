# Setting up Poetry and Virtual Environment

```shell
# update and upgrade packages
sudo apt update && sudo apt upgrade -y

# install necessary packages for python venv
sudo apt install -y python3-venv python3-pip

# create a virtual environment named 'venv'
python3 -m venv venv


# activate the virtual environment
source venv/bin/activate

# upgrade pip to the latest version
python -m pip install --upgrade pip

# Install Poetry
pip install poetry

# NOTE: copy pyproject.toml to current directory

# install packages from poetry.lock
poetry install --no-root


```

# Configuring Poetry Virtual Environment

```shell

# show poetry config
poetry config --list --local

# Create venv in project directory
poetry config virtualenvs.in-project true
poetry config virtualenvs.in-project true --local

# config poetry to default
poetry config virtualenvs.create false
poetry config virtualenvs.create false --local

# confirm venv location
poetry env info

# set poetry venv path
poetry config virtualenvs.create = true --local
poetry config virtualenvs.in-project = true --local
poetry config virtualenvs.options.always-copy = false --local
poetry config virtualenvs.options.no-pip = false --local
poetry config virtualenvs.options.system-site-packages = false --local
poetry config virtualenvs.path = "venv" --local
poetry config virtualenvs.prompt = "{project_name}-py{python_version}" --local
poetry config virtualenvs.use-poetry-python = false --local


# show poetry venvs
poetry env list

# poetry add venv with name
poetry add venv

# poetry rename venv
poetry env rename resource-py3.11 venv

# poetry remove venv
poetry env remove resource-eBqpJXeS-py3.11

# create venv using poetry
poetry install


# build wheel
poetry build

# poetry install from wheel
poetry install dist/*.whl

# Install from wheel
pip install dist/*.whl



```
