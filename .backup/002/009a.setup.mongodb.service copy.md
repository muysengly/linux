# Setup MongoDB Service on Ubuntu

```shell


# Verify system compatibility before installing MongoDB
lsb_release -a
uname -m
df -h
free -m


# Remove any existing MongoDB repositories
sudo rm -f /etc/apt/sources.list.d/mongodb-*.list
sudo apt update



# Import MongoDB GPG key
sudo mkdir -p /etc/apt/keyrings
wget -qO - https://www.mongodb.org/static/pgp/server-4.4.asc | sudo gpg --dearmor -o /etc/apt/keyrings/mongodb-org-4.4.gpg



# Create repository file for MongoDB 4.4
echo "deb [ arch=aarch64,amd64 signed-by=/etc/apt/keyrings/mongodb-org-4.4.gpg ] https://repo.mongodb.org/apt/debian buster/mongodb-org/4.4 main" | sudo tee /etc/apt/sources.list.d/mongodb-org-4.4.list





# Update package list
sudo apt update

# Install MongoDB 4.4.18
sudo apt install mongodb-org=4.4.18 mongodb-org-server=4.4.18 mongodb-org-shell=4.4.18 mongodb-org-mongos=4.4.18 mongodb-org-tools=4.4.18

# Hold the packages to prevent automatic updates
sudo apt-mark hold mongodb-org mongodb-org-server mongodb-org-shell mongodb-org-mongos mongodb-org-tools





# Update package list
sudo apt update

# Install MongoDB
sudo apt install -y mongodb-org

# aarch64

# No LSB modules are available.
# Distributor ID: Debian
# Description:    Debian GNU/Linux 13 (trixie)
# Release:        13
# Codename:       trixie



sudo apt-get update
sudo apt-get install -y wget gnupg
wget -qO - https://www.mongodb.org/static/pgp/server-4.4.asc | sudo tee /etc/apt/trusted.gpg.d/mongodb-server-4.4.gpg >/dev/null
echo "deb [ arch=amd64,arm64 ] https://repo.mongodb.org/apt/debian bullseye/mongodb-org/4.4 main" | sudo tee /etc/apt/sources.list.d/mongodb-org-4.4.list
sudo apt-get update
sudo apt-get install -y mongodb-org=4.4.18 mongodb-org-server=4.4.18 mongodb-org-shell=4.4.18 mongodb-org-mongos=4.4.18 mongodb-org-tools=4.4.18
sudo systemctl enable mongod
sudo systemctl start mongod
sudo systemctl status mongod
sudo apt-mark hold mongodb-org mongodb-org-server mongodb-org-shell mongodb-org-mongos mongodb-org-tools










```
