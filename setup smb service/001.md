```shell

# install samba server on ubuntu
sudo apt update && sudo apt upgrade -y
sudo apt install samba -y


# create or choose your user and set password
sudo adduser test_user
sudo adduser developer
sudo adduser deployer

# set samba password for the user
sudo smbpasswd -a test_user
sudo smbpasswd -a developer
# it will prompt you to enter and confirm the password


# enable the user
sudo smbpasswd -e test_user
sudo smbpasswd -e developer


# change smb password for the user
sudo smbpasswd test_user

# list all samba users
sudo pdbedit -L

# remove samba user
sudo smbpasswd -x developer


# prepare your shared folder
sudo mkdir -p /mnt/storage/test_user
sudo chown test_user:test_user /mnt/storage/test_user
sudo chmod 770 /mnt/storage/test_user


# configure Samba share
sudo nano /etc/samba/smb.conf


# At the bottom, add this block:
[smbshare]
   path = /mnt/storage/test_user
   browseable = yes
   read only = no
   valid users = test_user
   create mask = 0770
   directory mask = 0770
   force user = test_user
   force group = test_user



# You can change `[smbshare]` to any share name (e.g., `[storage]`).


sudo systemctl daemon-reexec
sudo systemctl daemon-reload
sudo systemctl enable smbd
sudo systemctl start smbd
sudo systemctl status smbd

# restart Samba service and check status
sudo systemctl restart smbd
sudo systemctl status smbd


# set maximum storage quota for test_user to 100MB
# sudo apt install quota -y
# sudo mount -o remount /mnt/storage
# sudo quotacheck -cum /mnt/storage
# sudo quotaon -v /mnt/storage

# set quota for user test_user to 100MB
sudo setquota -u test_user 100000 110000 0 0 /mnt/storage
# 100000 blocks = 100MB soft limit
# 110000 blocks = 110MB hard limit


# [OPTIONAL] allow Samba through UFW firewall
sudo ufw allow samba

```

## Windows:

1. Open **File Explorer**
2. In the address bar, type:

   \\your_server_ip\smbshare

3. Log in with:

   - **Username:** test_user
   - **Password:** (the one you set via `smbpasswd`)

## Linux:

smbclient //your_server_ip/smbshare -U test_user
