# Setup MongoDB

```shell

# update and upgrade packages
sudo apt update && sudo apt upgrade -y


# install required packages
sudo apt-get install gnupg curl -y


# import the public key used by the package management system
curl -fsSL https://www.mongodb.org/static/pgp/server-8.0.asc | sudo gpg -o /usr/share/keyrings/mongodb-server-8.0.gpg --dearmor

# create a list file for MongoDB
echo "deb [ arch=amd64,arm64 signed-by=/usr/share/keyrings/mongodb-server-8.0.gpg ] https://repo.mongodb.org/apt/ubuntu noble/mongodb-org/8.2 multiverse" | sudo tee /etc/apt/sources.list.d/mongodb-org-8.2.list

# install MongoDB
sudo apt-get update
sudo apt-get install mongodb-org -y

```

# Remove Mongodb Completly

```bash
## Step 1: Remove MongoDB Packages

# Stop MongoDB service first
sudo systemctl stop mongod
# Remove MongoDB packages
sudo apt purge mongodb-org* mongodb* mongodb-enterprise* -y
# Remove unused dependencies
sudo apt autoremove -y


## Step 2: Remove MongoDB Data and Logs

# Remove data directories
sudo rm -rf /var/lib/mongodb
sudo rm -rf /var/log/mongodb
sudo rm -rf /tmp/mongodb-*
# Remove configuration files
sudo rm -rf /etc/mongod.conf
sudo rm -rf /etc/mongodb.conf


## Step 3: Remove MongoDB Repository


# Remove MongoDB repository list
sudo rm -rf /etc/apt/sources.list.d/mongodb*.list
# Remove repository key
sudo apt-key del 41DE058A4E7DCA05
# Alternative method to remove the key file
sudo rm -rf /etc/apt/trusted.gpg.d/mongodb*.gpg
sudo rm -rf /usr/share/keyrings/mongodb*.gpg


## Step 4: Clean Up APT Cache

# Update package list to clear errors
sudo apt update
# Clean package cache
sudo apt clean
sudo apt autoclean


## Step 5: Verify Removal


# Check if any MongoDB packages remain
dpkg -l | grep mongo
# Check if MongoDB processes are running
ps aux | grep mongo
# Check if MongoDB directories still exist
ls -la /var/lib/ | grep mongo
ls -la /var/log/ | grep mongo


## Step 6: Remove MongoDB User (Optional)

# Remove MongoDB user if it exists
sudo userdel mongodb 2>/dev/null
sudo groupdel mongodb 2>/dev/null

```
