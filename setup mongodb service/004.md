# Setup MongoDB Service on Ubuntu

```shell

# update and upgrade packages
sudo apt update && sudo apt upgrade -y

# install required packages
sudo apt-get install gnupg curl -y


# import the public key used by the package management system
curl -fsSL https://www.mongodb.org/static/pgp/server-8.0.asc | sudo gpg -o /usr/share/keyrings/mongodb-server-8.0.gpg --dearmor

# remove
# sudo rm -rf /etc/apt/sources.list.d/mongodb-org-8.2.list


# create a list file for MongoDB
echo "deb [ arch=amd64,arm64 signed-by=/usr/share/keyrings/mongodb-server-8.0.gpg ] https://repo.mongodb.org/apt/ubuntu noble/mongodb-org/8.2 multiverse" | sudo tee /etc/apt/sources.list.d/mongodb-org-8.2.list

# remove everything of mongodb if needed
sudo apt-get purge mongodb-org*

# delete log and db folders
sudo rm -r /var/log/mongodb
sudo rm -r /var/lib/mongodb
sudo rm -rf /etc/mongod.conf
sudo rm -rf /etc/systemd/system/mongod.service

# reload local package database
sudo apt-get update




##########

# FIRST: Create admin user BEFORE enabling authorization
# Make sure authorization is NOT enabled in /etc/mongod.conf yet
# If it is enabled, comment it out, restart mongod, then proceed

# goto mongo shell (without authentication)
mongosh

# list all databases
show dbs

use admin
# set admin user password

use admin
db.auth("adminUser", "adminGTR")


mongosh -u "adminUser" -p "adminGTR" --authenticationDatabase "admin"

# set password for database database
use database
db.createUser(
  {
    user: "adminGTR",
    pwd: "adminGTR",
    roles: [ { role: "readWrite", db: "database" } ]
  }
)

# why other can write?
# db.grantRolesToUser("adminGTR", [ { role: "readWrite", db: "database" } ])



# THEN: Enable authorization in config
sudo nano /etc/mongod.conf

security:
    authorization: enabled

# reload mongodb
sudo systemctl daemon-reexec
sudo systemctl daemon-reload
sudo systemctl restart mongod
sudo systemctl status mongod

# Now authenticate with the created user
mongosh -u admin -p gtradmin --authenticationDatabase admin


# set admin password
use admin
db.changeUserPassword("admin", "gtradmin")


# remove mongodb service (if needed)
# sudo rm /etc/systemd/system/mongod.service





##########


# remove old list file if exists
# sudo rm -rf /etc/apt/sources.list.d/mongodb-org-8.0.list

# check debian version
lsb_release -dc

# how to know bullseye or buster
cat /etc/os-release

# reload local package database
sudo apt-get update

# install the MongoDB packages
sudo apt-get install mongodb-org -y

#
sudo systemctl daemon-reexec
sudo systemctl daemon-reload

# enable MongoDB service
sudo systemctl enable mongod
sudo systemctl start mongod
sudo systemctl status mongod

# reload mongodb
sudo systemctl daemon-reexec
sudo systemctl daemon-reload
sudo systemctl restart mongod
sudo systemctl status mongod

# stop mongodb
sudo systemctl stop mongod

# remove mongodb
sudo apt-get remove mongodb-org -y


# remove mongodb completely
sudo apt-get purge mongodb-org*
sudo rm -r /var/log/mongodb
sudo rm -r /var/lib/mongodb
sudo rm -rf /etc/mongod.conf
sudo rm -rf /etc/systemd/system/mongod.service



# list all serivices to verify mongod is running
sudo systemctl list-units --type=service --state=running
sudo systemctl list-units --type=service --all

# [TEST]

# create the MongoDB admin user
# mongo --eval 'db.getSiblingDB("admin").createUser({user: "admin", pwd: "jkl;'", roles: [{role: "userAdminAnyDatabase", db: "admin"}]})'

# mongosh --port 27017 -u admin -p "jkl;'"

# login to mongo shell
mongosh

# show all user in admin
use admin


# setup mongodb authentication
# use admin database
use admin


# show all databases
show dbs

show collections


# show all system users
show users


# list all runing services
sudo systemctl list-units --type=service --state=running
sudo systemctl list-units --type=service


# show location of mongodb.service for delete
sudo systemctl status mongod.service

# delete mongodb service (if needed)
# sudo rm /etc/systemd/system/mongod.service

# enable mongodb for specific user
sudo systemctl enable mongod --user


# config mongodb
sudo nano /etc/mongod.conf


# goto mongo shell
mongosh

# set permisions for specific database
use database

# check database roles
show roles

# remove permissions for user
db.dropUser("admin")

# set database passwrod
db.




db.createUser(
  {
    user: "admin",
    pwd: "admin",
    roles: [ { role: "readWrite", db: "database" } ]
  }
)

# exit mongo shell
exit


```

```shell
# Remove current installation
sudo systemctl stop mongod
sudo apt remove mongodb-org*  # or use yum/dnf

# Install MongoDB from OS repositories (usually better compatibility)
sudo apt update
sudo apt install mongodb

# Or for newer versions, use MongoDB's ARM64 repository
wget -qO - https://www.mongodb.org/static/pgp/server-6.0.asc | sudo apt-key add -
echo "deb [ arch=arm64 ] https://repo.mongodb.org/apt/ubuntu focal/mongodb-org/6.0 multiverse" | sudo tee /etc/apt/sources.list.d/mongodb-org-6.0.list
sudo apt update
sudo apt install mongodb-org



# remove mongodb from update
sudo apt-mark unhold mongodb-org mongodb-org-server mongodb-org-shell mongodb-org-mongos mongodb-org-tools




```
