```bash

#
sudo apt update && sudo apt install quota -y

# set maximum storage quota for test_user to 100MB
sudo apt install quota -y




sudo mount -o remount /mnt/storage
sudo quotacheck -cum /mnt/storage
sudo quotaon -v /mnt/storage





# set quota for user test_user to 100MB
sudo setquota -u test_user 100000 110000 0 0 /mnt/storage
# 100000 blocks = 100MB soft limit
# 110000 blocks = 110MB hard limit




# set quota for user developer to 500MB
sudo setquota -u developer 500000 550000 0 0 /home/developer

# set quota for user deployer to 500MB
sudo setquota -u deployer 500000 550000 0 0 /home/deployer


# [OPTIONAL] allow Samba through UFW firewall
sudo ufw allow samba




```
