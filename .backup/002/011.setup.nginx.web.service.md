# Setup Flutter Web for Ubuntu

```shell

# update and upgrade packages
sudo apt update && sudo apt upgrade -y

# install nginx to serve the built web app
sudo apt install nginx -y

# check nginx status
sudo systemctl status nginx

# create directory to host flutter web app
sudo mkdir -p /var/www/web

# set ownership to current user
# sudo chown -R root:root /var/www
sudo chown -R $USER:$USER /var/www


# create Nginx configuration for Flutter web app
sudo nano /etc/nginx/sites-available/web

# rename from /etc/nginx/sites-available/flutter_web to /etc/nginx/sites-available/web
# sudo mv /etc/nginx/sites-available/flutter_web /etc/nginx/sites-available/web

# [ADD]
server {
    listen 80;
    server_name gtr.codeshift.me;
    root /var/www/web;
    index index.html;
    location / {
        try_files $uri $uri/ /index.html;
    }
}



# enable the new site and disable the default site
sudo ln -s /etc/nginx/sites-available/web /etc/nginx/sites-enabled/web


# disable default site
sudo rm /etc/nginx/sites-enabled/default

# remove old default site configuration
sudo rm /etc/nginx/sites-enabled/web

# Test Nginx configuration
sudo nginx -t



# reload Nginx to apply changes
# sudo systemctl reload nginx
sudo systemctl restart nginx
sudo systemctl status nginx


# stop nginx service
sudo systemctl stop nginx
sudo systemctl status nginx

# remove nginx
sudo systemctl stop nginx
sudo apt remove nginx -y
sudo apt purge nginx -y
sudo rm /etc/nginx/sites-available/web
sudo rm /etc/nginx/sites-enabled/web
sudo rm -rf /var/www/web



```

```shell

# ----------

# Run in Chrome with debug mode
flutter run -d chrome --debug

# Run in release mode
flutter run -d chrome --release

# Build for production
flutter build web --release

# Serve built files locally
cd build/web
python3 -m http.server 8000

# [OPTIONAL] Install Android Studio for mobile development
sudo snap install android-studio --classic

# [OPTIONAL] install flutter via snap
sudo snap install flutter --classic

# remove flutter snap if already installed
sudo snap remove flutter --classic

# Set up flutter for web development
flutter channel stable
flutter upgrade
flutter config --enable-web
flutter doctor

# completly remove flutter
sudo rm -rf ~/flutter

# stop nginx service and remove completely
sudo systemctl stop nginx
sudo apt remove nginx -y


# remove nginx configuration files
sudo rm /etc/nginx/sites-available/web
sudo rm /etc/nginx/sites-enabled/web
sudo rm -rf /var/www/web

# uninstall nginx
sudo apt purge nginx -y





```
