# Setup Minio Service on Ubuntu

```shell
# update and prepare your system
sudo apt update && sudo apt upgrade -y

# install wget if not already installed
sudo apt install wget -y

# download the latest MinIO server binary
wget https://dl.min.io/server/minio/release/linux-amd64/minio

# make it executable
sudo chmod +x minio

# move it to /usr/local/bin for global access
sudo mv minio /usr/local/bin/

# create a directory for MinIO data storage
sudo mkdir -p /mnt/storage/minio

# set ownership to root (or a specific user if needed)
sudo chown -R root:root /mnt/storage/minio


# create a systemd service for MinIO
sudo nano /etc/systemd/system/minio.service

# paste the following content into the file:

# ---------- minio.service ---------
[Unit]
Description=MinIO Object Storage
Documentation=https://min.io/docs/
Wants=network-online.target
After=network-online.target

[Service]
User=root
Group=root
Environment="MINIO_ROOT_USER=gtr"
Environment="MINIO_ROOT_PASSWORD=qwertyui"
ExecStart=/usr/local/bin/minio server /mnt/storage/minio --console-address ":9001"
Restart=always
LimitNOFILE=65536

[Install]
WantedBy=multi-user.target
# ---------- minio.service ---------

# NOTE: Change MINIO_ROOT_USER and MINIO_ROOT_PASSWORD to secure values


# # reload systemd to recognize the new service
sudo systemctl daemon-reload
sudo systemctl enable minio
sudo systemctl start minio
sudo systemctl status minio



# [OPTIONAL] If UFW is enabled, open MinIO ports
sudo ufw allow 9000
sudo ufw allow 9001
sudo ufw reload



# uninstall MinIO service
sudo systemctl stop minio

sudo systemctl disable minio

sudo rm /etc/systemd/system/minio.service

sudo systemctl daemon-reload

# remove MinIO binary
sudo rm /usr/local/bin/minio

# remove MinIO data directory
sudo rm -rf /mnt/storage/minio

# verify removal
sudo systemctl status minio

```
