# 1. setup ubuntu

```shell
# update and upgrade packages
sudo apt update && sudo apt upgrade -y

# install necessary packages
sudo apt install curl -y



# create a new user with home directory
sudo adduser bunhap

# add user to sudo group
sudo usermod -aG sudo bunhap

# add user password
sudo passwd bunhap

# switch to new user
su - bunhap


# add access to mysql with specific database
sudo usermod -aG mysql bunhap



```

# 1.1. configure ubuntu power settings (no sleep/suspend)

```shell
# disable automatic suspend
sudo systemctl mask sleep.target suspend.target hibernate.target hybrid-sleep.target

# modify logind.conf to ignore lid switch
sudo sed -i 's/#HandleLidSwitch=suspend/HandleLidSwitch=ignore/' /etc/systemd/logind.conf
sudo sed -i 's/#HandleLidSwitchExternalPower=suspend/HandleLidSwitchExternalPower=ignore/' /etc/systemd/logind.conf

# restart systemd-logind service
sudo systemctl restart systemd-logind

# verify power settings
systemctl status sleep.target


# Manually:
# sudo nano /etc/systemd/logind.conf

# CHANGE TO:
# HandleLidSwitch=ignore
# HandleLidSwitchExternalPower=ignore
# HandleLidSwitchDocked=ignore

```

# 2. install openssh-server

```shell
# install openssh-server
sudo apt install openssh-server -y

# configure ssh to allow port 22 an d
sudo nano /etc/ssh/sshd_config

# uncomment and set the following lines:
# Port 22
# ListenAddress 0.0.0.0


# check ssh service status
sudo systemctl enable ssh
sudo systemctl start ssh
sudo systemctl status ssh



```

# 5. setup mysql server

```shell
# install mysql server
sudo apt install mysql-server -y

# secure mysql installation
sudo mysql_secure_installation

# login to mysql
sudo mysql -u root -p




# set root password
ALTER USER 'root'@'localhost' IDENTIFIED WITH mysql_native_password BY 'qwertyui';
FLUSH PRIVILEGES;


# -- change password for an existing MySQL user
ALTER USER 'username'@'localhost' IDENTIFIED WITH mysql_native_password BY 'NewStrongP@ssw0rd!';
FLUSH PRIVILEGES;

ALTER USER 'username'@'localhost' IDENTIFIED WITH mysql_native_password BY 'NewStrongP@ssw0rd!';
FLUSH PRIVILEGES;



# add user and grant privileges
CREATE USER 'bunhap'@'localhost' IDENTIFIED BY 'NewStrongP@ssw0rd!';
GRANT ALL PRIVILEGES ON db_bunhap_dev.* TO 'bunhap'@'localhost';
```
