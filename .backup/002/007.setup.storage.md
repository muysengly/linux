# Setup Storage Server on Ubuntu

```shell


# list all hard disks
lsblk

# format new disk (e.g., /dev/sdb) with ext4 filesystem
sudo mkfs.ext4 /dev/sda
sudo mkfs.ext4 /dev/sdb

# create a mount point
sudo mkdir -p /mnt/storage

# ----------

# setup main storage disk
# mount the disk to the mount point
sudo mount /dev/sda /mnt/storage

# setup sdb as backup storage for user data
# create a mount point for backup storage
sudo mkdir -p /mnt/backup_storage

# mount the backup disk to the mount point
sudo mount /dev/sdb /mnt/backup_storage







# prepare backup storage folder
# create backup storage folder
# sudo mkdir -p /mnt/backup_storage
# set ownership to specific user (e.g., $USER)
sudo chown root:root /mnt/backup_storage
# sudo chown $USER:$USER /mnt/backup_storage
sudo chown root:root /mnt/storage
# sudo chown $USER:$USER /mnt/storage
# set permissions
sudo chmod 755 /mnt/backup_storage
sudo chmod 755 /mnt/storage
# dont let other user out off root to access backup folder
sudo chmod o-rwx /mnt/backup_storage


# test `rsync` manually
sudo rsync -av --delete /mnt/storage/ /mnt/backup_storage/
# sudo rsync -a --delete --exclude='lost+found' --exclude='aquota.user' /mnt/storage/ /mnt/backup_storage/
# -a: archive mode (preserves permissions, timestamps, symbolic links, etc.)
# -v: verbose output
# --delete: delete files in the destination that are no longer present in the source


# open crontab for editing
sudo crontab -e
# add the following line to run rsync every day at 2am and log output
0 2 * * * rsync -a --delete /mnt/storage/ /mnt/backup_storage/ >> /var/log/backup.log 2>&1
# [NOTE]: the log file `/var/log/backup.log` will contain the output of the backup process.
# 0 2 * * *: schedule the job to run daily at 2:00 AM


# [OPTIONAL] - Create compressed archive backups
tar -czpf /mnt/backup_storage/storage_$(date +%F).tar.gz -C /mnt storage
# -c: create a new archive
# -z: compress the archive using gzip
# -p: preserve permissions
# -f: specify the filename of the archive
# -C /mnt: change to the /mnt directory before creating the archive
# storage: the directory to be archived













# [OPTIONAL] configure fstab for automatic mounting on boot
echo '/dev/sda /mnt/storage ext4 defaults 0 2' | sudo tee -a /etc/fstab
echo '/dev/sdb /mnt/backup_storage ext4 defaults 0 2' | sudo tee -a /etc/fstab



# ----------



# install quota package for disk quota management
sudo apt install quota -y


# remount the filesystem with quota options
sudo systemctl daemon-reload
sudo mount -o remount /mnt/storage

# create quota files and turn on quota
sudo quotacheck -cum /
sudo quotaon -v /
# for storage mount point
sudo quotacheck -cum /mnt/storage
sudo quotaon -v /mnt/storage
# for backup storage mount point
sudo quotacheck -cum /mnt/backup_storage
sudo quotaon -v /mnt/backup_storage



# check main filesystem quota status
df -h /
df -h /mnt/storage
df -h /mnt/backup_storage







# [OPTIONAL] setup user quota

# create user
sudo adduser bunhap

# set quota for user bunhap on /home/bunhap to 2GB
sudo setquota -u bunhap 2000000 2200000 0 0


```
