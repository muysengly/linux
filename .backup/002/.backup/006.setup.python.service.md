# Setup Service on Ubuntu

```shell

# create GTR service
sudo nano /etc/systemd/system/gtr_backend.service


[Unit]
Description='GTR Backend Service'
After=network.target

[Service]
User=root
WorkingDirectory=/home/gtr/gtr_backend/
ExecStart=/bin/bash -c 'source /home/gtr/gtr_backend/venv/bin/activate && python service_main.py'
Restart=always

[Install]
WantedBy=multi-user.target

# reload the systemd daemon to recognize the new service file
sudo systemctl daemon-reexec
sudo systemctl daemon-reload

# enable and start a service
sudo systemctl enable gtr_backend
sudo systemctl start gtr_backend
sudo systemctl status gtr_backend

# reload gtr_backend service
sudo systemctl restart gtr_backend
sudo systemctl status gtr_backend

# stop gtr_backend service
sudo systemctl disable gtr
sudo systemctl stop gtr

# remove gtr service
sudo rm /etc/systemd/system/gtr.service

```

# create front-end service

```shell

# create gtr_frontend directory
mkdir -p ~/gtr_frontend

# set permissions
sudo chown -R gtr:gtr ~/gtr_frontend

# initialize python venv
python3 -m venv ~/gtr_frontend/venv

# activate venv
source ~/gtr_frontend/venv/bin/activate

# install required packages
pip install numpy
pip install fastapi
pip install uvicorn
pip install httpx
pip install jinja2


# create GTR service
sudo nano /etc/systemd/system/gtr_frontend.service


[Unit]
Description='GTR Frontend Service'
After=network.target

[Service]
User=root
WorkingDirectory=/home/gtr/gtr_frontend/
ExecStart=/usr/bin/python3 -m http.server 80 --directory web --bind localhost
Restart=always

[Install]
WantedBy=multi-user.target


# reload the systemd daemon to recognize the new service file
sudo systemctl daemon-reexec
sudo systemctl daemon-reload


# enable and start a service
sudo systemctl enable gtr_frontend
sudo systemctl start gtr_frontend
sudo systemctl status gtr_frontend


# reload gtr_frontend service
sudo systemctl restart gtr_frontend
sudo systemctl status gtr_frontend


# stop gtr_frontend service
sudo systemctl disable gtr_frontend
sudo systemctl stop gtr_frontend
```
